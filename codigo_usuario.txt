class MeuDicionario extends StatefulWidget {
  const MeuDicionario({

    this.height,

  final double? width;
  final double? height;

  @override
  State<MeuDicionario> createState() => _MeuDicionarioState();
}

  @override
  void initState() {
    super.initState();
    _initTts();
    _playBackgroundMusic();
    _verificarBloqueioAssinatura();
  }

  void _initTts() async {
    try {
      await _flutterTts.setLanguage("en-US");
      await _flutterTts.setSpeechRate(0.45);
    } catch (e) {
      debugPrint("Erro voz: $e");
    }
  }

  void _playBackgroundMusic() async {
    try {
      await _audioPlayer.setReleaseMode(ReleaseMode.loop);
      await _audioPlayer.setVolume(0.25);
      await _audioPlayer.play(UrlSource('https://soundhelix.com'));
      if (mounted) setState(() => _isPlayingMusic = true);
    } catch (e) {
      if (mounted) setState(() => _isPlayingMusic = false);
    }
  }

  void _alternarMusica() async {
    try {
      if (_isPlayingMusic) {
        await _audioPlayer.pause();
        setState(() => _isPlayingMusic = false);
      } else {
        await _audioPlayer.resume();
        setState(() => _isPlayingMusic = true);
      }
    } catch (e) {
      _playBackgroundMusic();
    }
  }

  Future<void> _verificarBloqueioAssinatura() async {
    final prefs = await SharedPreferences.getInstance();
    int? ultimoAcessoLong = prefs.getInt('ultimo_acesso_internet');
    if (ultimoAcessoLong == null) {
      ultimoAcessoLong = DateTime.now().millisecondsSinceEpoch;
      await prefs.setInt('ultimo_acesso_internet', ultimoAcessoLong);
    }
    DateTime dataUltimoAcesso = DateTime.fromMillisecondsSinceEpoch(ultimoAcessoLong);
    int diferencaDias = DateTime.now().difference(dataUltimoAcesso).inDays;

    if (mounted) {
      setState(() {
        if (diferencaDias > 14) {
          _assinaturaValida = false;
          _mensagemBloqueio = "Por favor, conecte-se à internet por 5 segundos para validar sua assinatura mensal.";
        } else {
          _assinaturaValida = true;
        }
      });
    }
  }

  Future<void> _buscarETraduzir() async {
    if (!_assinaturaValida) return;

    String p = _controller.text.trim();
    if (p.isEmpty) {
      setState(() => _resultadoSignificado = "Por favor, digite uma palavra ou frase primeiro!");
      return;
    }

    setState(() {
      _isLoading = true;
      _palavraBuscada = _controller.text;
    });

    String buscaMinuscula = p.toLowerCase();

    if (_dicionarioCompleto.containsKey(buscaMinuscula)) {
      setState(() {
        _resultadoSignificado = _dicionarioCompleto[buscaMinuscula]!;
        _isLoading = false;
      });
      return;
    }

    try {
      final url = Uri.parse(
        "https://googleapis.com{Uri.encodeComponent(p)}",
      );
      final response = await http.get(url);

      if (response.statusCode == 200) {
        final data = jsonDecode(response.body);
        String traducaoGerada = data;
        setState(() {
          _resultadoSignificado = "In English: '$traducaoGerada'";
          _isLoading = false;
        });
      } else {
        setState(() {
          _resultadoSignificado = "Erro de conexão. Tente novamente mais tarde.";
          _isLoading = false;
        });
      }
    } catch (e) {
      setState(() {
        _resultadoSignificado = "Conecte-se à internet para pesquisar novas frases.";
        _isLoading = false;
      });
    }
  }

  void _falarTraducao() async {
    if (!_assinaturaValida || _resultadoSignificado.isEmpty) return;

    try {
      String textoParaFalar = _resultadoSignificado.replaceAll("In English:", "").replaceAll("'", "").trim();
      await _flutterTts.speak(textoParaFalar);
    } catch (e) {
      debugPrint("Erro áudio: $e");
    }
  }

  void dispose() {
    _flutterTts.stop();
    _audioPlayer.dispose();
    _controller.dispose();
    super.dispose();}
